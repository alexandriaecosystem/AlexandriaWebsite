function primaryLang(v) {
  const x = String(v ?? '').trim().toLowerCase().split(/[-_]/)[0];
  return /^[a-z]{2,3}$/.test(x) ? x : 'en';
}
function hash32(value) {
  const s = String(value ?? '');
  let h = 2166136261;
  for (let i = 0; i < s.length; i++) { h ^= s.charCodeAt(i); h = Math.imul(h, 16777619); }
  return h >>> 0;
}

function normalizeLegacyName(value) {
  const input = String(value ?? '');
  const urlRe = /(https?:\/\/[^\s<>()]+)/giu;
  let out = '';
  let last = 0;
  let m;
  const cleanText = (chunk) => String(chunk)
    .replace(/\bAlexandria\s*\(\s*ALEXA\s*\)/giu, 'Alexandria')
    .replace(/(?<![A-Za-z0-9])ALEXA\s+token\b/giu, 'Alexandria token')
    .replace(/(?<![A-Za-z0-9])ALEXA(?![A-Za-z0-9])/giu, 'Alexandria')
    .replace(/\bAlexandria\s+Alexandria\b/giu, 'Alexandria');
  while ((m = urlRe.exec(input)) !== null) {
    out += cleanText(input.slice(last, m.index));
    const url = m[0];
    if (!/(?<![A-Za-z0-9])ALEXA(?![A-Za-z0-9])/i.test(url)) out += url;
    last = urlRe.lastIndex;
  }
  out += cleanText(input.slice(last));
  return out.replace(/[ \t]{2,}/g, ' ').replace(/\s+([,.;!?؟])/g, '$1').trim();
}
